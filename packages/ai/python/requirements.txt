# rembg is pinned to 2.0.69. The patched 2.0.75 (GHSA: server SSRF/CORS +
# custom-model path traversal) pulls numpy>=2.3, scipy>=1.16 and
# scikit-image>=0.26 -- a numpy 2.x closure that is incompatible with this
# numpy==1.26.4-locked stack (realesrgan 0.3.0 and codeformer-pip 0.0.4 break
# on numpy 2.x). Both advisories are unreachable here: rembg is used purely as
# a library (we never run `rembg s`), and new_session() only ever receives
# allowlisted model names (remove_bg.py ALLOWED_MODELS), never user paths.
rembg[cpu]==2.0.69
realesrgan==0.3.0
paddleocr[doc-parser]>=3.4.0,<3.5.0
paddlepaddle>=3.0.0,<3.1.0
# mediapipe is split by CPU arch. PyPI has no aarch64 wheel above 0.10.18, and
# 0.10.18 calls MessageFactory.GetPrototype (removed in protobuf 5+), so on arm64
# it must run with protobuf<5. Left unconstrained, the paddle/onnxruntime deps pull
# protobuf 7.x into the shared /data/ai/venv and silently break mediapipe's
# FaceLandmarker: red-eye-removal fails on every input (blur-faces/smart-crop keep
# working since face detection loads a prebuilt graph and skips GetPrototype).
# x86_64 has mediapipe 0.10.35, which works with protobuf 7, so it stays unpinned.
mediapipe>=0.10.21; platform_machine != "aarch64"
mediapipe>=0.10.18,<0.10.21; platform_machine == "aarch64"
protobuf>=4.25.3,<5; platform_machine == "aarch64"
onnxruntime==1.20.1
numpy==1.26.4
Pillow==12.2.0
opencv-python-headless==4.10.0.84
codeformer-pip==0.0.4
